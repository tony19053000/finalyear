if "qos" in sys.argv:   # HighQoS: tell Windows "do NOT throttle / do not prefer efficiency cores"

cores = set()

W=32; r=np.random.default_rng(1)
w1=r.integers(-8,8,(27,W)); w2=r.integers(-8,8,(9*W,W)); w3=r.integers(-8,8,(W,10))
def im2col(x):
    s=x.shape[1]; p=np.pad(x,((0,0),(1,1),(1,1),(0,0))); return np.concatenate([p[:,a:a+s,b:b+s,:] for a in range(3) for b in range(3)],axis=-1)
def pool(x): return (x[:,0::2,0::2]+x[:,1::2,0::2]+x[:,0::2,1::2]+x[:,1::2,1::2])>>2
